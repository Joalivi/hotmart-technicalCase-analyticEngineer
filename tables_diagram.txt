table purchase{
  purchase_id bigint [note: 'identificador da compra']
  buyer_id bigint [note: 'identificador do comprador']
  prod_item_id bigint [note: 'identificador do item de compra']
  order_date date [note: 'data do pedido de compra']
  release_date date [note: 'data de liberação da compra mediante a confirmação do pagamento']
  producer_id bigint [note: 'identificador do produtor']
  purchase_partition bigint [note: 'partição no lake para a compra']
  prod_item_partition bigint [note: 'partição no lake para o item de compra']
  purchase_total_value float [note: 'valor total da compra']
  purchase_status string [note: 'status da compra: INICIADA, APROVADA, CANCELADA, REEMBOLSADA']
  transaction_datetime datetime [note: 'momento de inserção do dado no lake']
  transaction_date date [note: 'data de inserção do dado no lake']
}
